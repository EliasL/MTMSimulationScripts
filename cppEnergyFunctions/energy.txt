// ContiEnergy energy density; generated from MTMath.energyFunction.
double energyDensity(double C11, double C22, double C12, double beta, double K, double noise) {
  double x0 = C11*C22 - pow(C12, 2);
  double x1 = noise*x0;
  double x2 = pow(x0, -1.0/2.0);
  double x3 = C11*x2;
  double x4 = C22*x2;
  double x5 = pow(x3 - x4, 2);
  double x6 = C12*x2;
  double x7 = x3 + x4;
  double x8 = -4*x6 + x7;
  double x9 = pow(x8, 2);
  double x10 = pow(x5 + 0.33333333333333331*x9, 3);
  double x11 = x5*x8 - 0.1111111111111111*pow(x8, 3);
  double x12 = pow(x11, 2);
  double x13 = -x6 + x7;
  double x14 = 0.25*x5 + 0.083333333333333329*x9;
  double x15 = x11*x14;

  double phi = K*(x1 - log(x1)) + beta*(-0.0064709595959595969*x10 + 0.000946969696969697*x12 + 0.012345679012345677*pow(x13, 4)*x14 + 0.035353535353535352*x15*(x3 + x4 - x6)) + 0.005681818181818182*x10 + 0.037037037037037028*x11*pow(x13, 3) + 0.032196969696969696*x12 - 0.24242424242424243*x15*(x3 + x4 - x6);

  return phi;
}
